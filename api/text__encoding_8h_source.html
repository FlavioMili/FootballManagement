49inline void appendFixed(std::string& out, float value, int decimals)
50{
51 constexpr std::array<std::int64_t, 5> SCALES = {1, 10, 100, 1000, 10000};
52 const std::int64_t scale = SCALES[static_cast<std::size_t>(decimals)];
53 // float * 10^k is exact in double; round() is half away from zero.
54 const double scaled =
55 std::isfinite(value)
56 ? std::round(static_cast<double>(value) * static_cast<double>(scale))
57 : 0.0;
58 const auto units = static_cast<std::int64_t>(std::fabs(scaled));
59 if (scaled < 0.0) out.push_back('-');
60 appendInt(out, units / scale);
61 if (decimals == 0) return;
62 out.push_back('.');
63 std::int64_t fraction = units % scale;
64 std::array<char, 4> digits{};
65 for (int i = decimals - 1; i >= 0; --i)
66 {
67 digits[static_cast<std::size_t>(i)] =
68 static_cast<char>('0' + fraction % 10);
69 fraction /= 10;
70 }
71 out.append(digits.data(), static_cast<std::size_t>(decimals));
72}

24inline ImVec4 ratingColor(double rating)
25{
26 if (rating >= 70.0) return ImVec4(0.25f, 0.78f, 0.48f, 1.0f);
27 if (rating >= 55.0) return ImVec4(0.95f, 0.72f, 0.25f, 1.0f);
28 return ImVec4(0.92f, 0.38f, 0.36f, 1.0f);
29}

44inline void detailPanel(const char* id, const Player* player,
45 const StatsConfig& statsConfig,
46 const Player* comparison = nullptr, float height = 0.0f)
47{
48 ImGui::BeginChild(id, ImVec2(0.0f, height), true);
49 if (!player)
50 {
51 ImGui::PushStyleColor(ImGuiCol_Text,
52 ImGui::GetStyleColorVec4(ImGuiCol_TextDisabled));
53 ImGui::TextWrapped("%s", LOC("PLAYER_SELECT_PROMPT"));
54 ImGui::PopStyleColor();
55 ImGui::EndChild();
56 return;
57 }
58
59 const double overall = player->getOverall(statsConfig);
60 ImGui::TextColored(ratingColor(overall), "%.1f", overall);
61 ImGui::SameLine();
62 ImGui::Text("%s", player->getName().c_str());
63 ImGui::PushStyleColor(ImGuiCol_Text,
64 ImGui::GetStyleColorVec4(ImGuiCol_TextDisabled));
65 ImGui::TextWrapped("%s | %s %d | %d cm | %s",
66 RoleUtils::toString(player->getRole()).c_str(),
67 LOC("PLAYER_AGE"), player->getAge(), player->getHeight(),
68 player->getFoot() == Foot::Right
69 ? LOC("PLAYER_FOOT_RIGHT")
70 : LOC("PLAYER_FOOT_LEFT"));
71 ImGui::PopStyleColor();
72 ImGui::Separator();
73 ImGui::Text("%s: %u", LOC("PLAYER_WEEKLY_WAGE"), player->getWage());
74 const ImVec4 contractColor = player->getContractYears() <= 1
75 ? ImVec4(0.92f, 0.38f, 0.36f, 1.0f)
76 : ImVec4(0.45f, 0.65f, 0.85f, 1.0f);
77 ImGui::TextColored(contractColor, "%s: %u", LOC("PLAYER_CONTRACT"),
78 static_cast<unsigned>(player->getContractYears()));
79
80 if (comparison)
81 {
82 const double delta = overall - comparison->getOverall(statsConfig);
83 ImGui::TextColored(delta >= 0.0 ? ImVec4(0.25f, 0.78f, 0.48f, 1.0f)
84 : ImVec4(0.92f, 0.38f, 0.36f, 1.0f),
85 "%s: %+.1f", LOC("PLAYER_OVERALL_CHANGE"), delta);
86 if (player->getRole() == comparison->getRole())
87 {
88 ImGui::TextColored(ImVec4(0.25f, 0.78f, 0.48f, 1.0f), "%s",
89 LOC("PLAYER_ROLE_MATCH"));
90 }
91 else
92 {
93 ImGui::TextColored(ImVec4(0.95f, 0.72f, 0.25f, 1.0f), "%s: %s -> %s",
94 LOC("PLAYER_ROLE_CHANGE"),
95 RoleUtils::toString(comparison->getRole()).c_str(),
96 RoleUtils::toString(player->getRole()).c_str());
97 }
98 }
99
100 ImGui::SeparatorText(LOC("PLAYER_ATTRIBUTES"));
101 for (const auto& [name, value] : player->getStats())
102 {
103 const float normalized = std::clamp(value / 100.0f, 0.0f, 1.0f);
104 const float comparisonValue =
105 comparison && comparison->getStats().contains(name)
106 ? comparison->getStats().at(name)
107 : 0.0f;
108 const float delta = value - comparisonValue;
109 const std::string valueText = std::format("{:.0f}", value);
110 const std::string deltaText = std::format("{:+.0f}", delta);
111 const float valueWidth = ImGui::CalcTextSize(valueText.c_str()).x;
112 const float deltaWidth =
113 comparison ? ImGui::CalcTextSize(deltaText.c_str()).x : 0.0f;
114 const float textSpacing =
115 comparison ? ImGui::GetStyle().ItemSpacing.x : 0.0f;
116 ImGui::TextUnformatted(name.c_str());
117 const float rightAlignedX = ImGui::GetCursorPosX() +
118 ImGui::GetContentRegionAvail().x - valueWidth -
119 deltaWidth - textSpacing;
120 ImGui::SameLine(std::max(ImGui::GetCursorPosX(), rightAlignedX));
121 ImGui::TextColored(ratingColor(value), "%s", valueText.c_str());
122 if (comparison)
123 {
124 ImGui::SameLine();
125 ImGui::TextColored(delta >= 0.0f ? ImVec4(0.25f, 0.78f, 0.48f, 1.0f)
126 : ImVec4(0.92f, 0.38f, 0.36f, 1.0f),
127 "%s", deltaText.c_str());
128 }
129 ImGui::PushStyleColor(ImGuiCol_PlotHistogram, ratingColor(value));
130 ImGui::ProgressBar(normalized, ImVec2(-1.0f, 14.0f), "");
131 ImGui::PopStyleColor();
132 }
133 ImGui::EndChild();
134}

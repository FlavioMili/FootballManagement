70inline MatchRenderSnapshot buildMatchRenderSnapshot(const MatchEngine& engine)
71{
72 MatchRenderSnapshot snapshot;
73 const auto& players = engine.getPlayers();
74 const auto& previousPositions = engine.getPreviousPlayerPositions();
75 const auto& previousFacingAngles = engine.getPreviousPlayerFacingAngles();
76 snapshot.players.reserve(players.size());
77 for (std::size_t index = 0; index < players.size(); ++index)
78 {
79 const MatchPlayer& source = players[index];
80 MatchRenderPlayer& renderPlayer = snapshot.players.emplace_back();
81 renderPlayer.player = source.player;
82 renderPlayer.isHomeTeam = source.isHomeTeam;
83 renderPlayer.currentPosition = source.position;
84 renderPlayer.movementTarget = source.movementTarget;
85 renderPlayer.intent = source.intent;
86 renderPlayer.currentFacingAngle = source.facingAngle;
87 renderPlayer.stamina = source.stamina;
88 renderPlayer.possessesBall = source.player != nullptr &&
89 engine.getBall().possessedBy == source.player;
90 renderPlayer.previousPosition = index < previousPositions.size()
91 ? previousPositions[index]
92 : renderPlayer.currentPosition;
93 renderPlayer.previousFacingAngle = index < previousFacingAngles.size()
94 ? previousFacingAngles[index]
95 : renderPlayer.currentFacingAngle;
96 }
97
98 const MatchBall& ball = engine.getBall();
99 snapshot.ball.currentPosition = ball.position;
100 snapshot.ball.previousPosition = engine.getPreviousBallPosition();
101 snapshot.ball.currentZ = ball.z;
102 snapshot.ball.previousZ = engine.getPreviousBallZ();
103 snapshot.ball.possessedBy = ball.possessedBy;
104
105 snapshot.state = engine.getState();
106 snapshot.homePhase = engine.getHomePhase();
107 snapshot.awayPhase = engine.getAwayPhase();
108 snapshot.transitionSecondsRemaining = engine.getTransitionSecondsRemaining();
109 snapshot.homeScore = engine.getHomeScore();
110 snapshot.awayScore = engine.getAwayScore();
111 snapshot.goalScoredByHome = engine.getGoalScoredByHome();
112 snapshot.goalCelebrationRemaining = engine.getGoalCelebrationRemaining();
113 snapshot.matchTimeMinutes = engine.getMatchTimeMinutes();
114 snapshot.events = &engine.getEvents();
115 snapshot.stats = &engine.getStats();
116 snapshot.interpolationAlpha = engine.getInterpolationAlpha();
117 return snapshot;
118}
